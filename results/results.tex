\begin{frame}{Key Takeaways}
    \begin{itemize}\setlength\itemsep{0.8em}
        \item Sessions and refinements shape client-side performance
        \item Random sequences and micro benchmarks miss this
        \item[$\Rightarrow$] SolidSessionBench: realistic, reproducible query sequences
    \end{itemize}

    \vspace{1.5em}
    \textbf{Future work}
    \begin{itemize}\setlength\itemsep{0.4em}
        \item Session-aware caching: sizing, indexing, and eviction tuned to sessions
        \item Reusing previous executions for query planning
    \end{itemize}
\end{frame}
